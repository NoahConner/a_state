<VirtualHost *:443>
    ServerName astateinsurance.com

    SSLEngine on
    SSLCertificateFile /etc/letsencrypt/live/astateinsurance.com/fullchain.pem
    SSLCertificateKeyFile /etc/letsencrypt/live/astateinsurance.com/privkey.pem
    Include /etc/letsencrypt/options-ssl-apache.conf

    ProxyPreserveHost On

# Add these two lines
    RequestHeader set X-Forwarded-Proto "https"
    RequestHeader set X-Forwarded-Host "astateinsurance.com"

    ProxyPass / http://localhost:4000/
    ProxyPassReverse / http://localhost:4000/

    ErrorLog ${APACHE_LOG_DIR}/astateinsurance-error.log
    CustomLog ${APACHE_LOG_DIR}/astateinsurance-access.log combined
</VirtualHost>

<VirtualHost *:443>
    ServerName www.astateinsurance.com

    SSLEngine on
    SSLCertificateFile /etc/letsencrypt/live/astateinsurance.com/fullchain.pem
    SSLCertificateKeyFile /etc/letsencrypt/live/astateinsurance.com/privkey.pem
    Include /etc/letsencrypt/options-ssl-apache.conf

    RewriteEngine On
    RewriteRule ^ https://astateinsurance.com%{REQUEST_URI} [R=301,L]
</VirtualHost>
